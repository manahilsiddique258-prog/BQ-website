const reviews = [
    {
        text: '"Learning new technology skills has helped me become more confident and explore new opportunities."',
        name: "Student Experience"
    },
    {
        text: '"Practical learning and discovering new digital skills can make a real difference."',
        name: "Learning Journey"
    },
    {
        text: '"Every new skill is another step towards building a better future."',
        name: "Student Motivation"
    }
];

let reviewIndex = 0;

function nextReview() {
    reviewIndex = (reviewIndex + 1) % reviews.length;

    document.getElementById("reviewText").textContent =
        reviews[reviewIndex].text;

    document.getElementById("reviewName").textContent =
        reviews[reviewIndex].name;
}

// CONTACT FORM

document.getElementById("contactForm").addEventListener("submit", function(e) {
    e.preventDefault();

    const name = document.getElementById("name").value.trim();
    const email = document.getElementById("email").value.trim();
    const course = document.getElementById("course").value;
    const message = document.getElementById("message").value.trim();

    if (!name || !email || !course || !message) {
        alert("Please fill in all fields.");
        return;
    }

    alert("Thank you, " + name + "! Your form has been validated. Connect a backend or email service to receive submissions.");

    this.reset();
});
